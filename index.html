import random from flask import Flask, render_template_string, request app = Flask(__name__) # Giao diện HTML của trang web (Nút bấm và màu sắc) HTML_INTERFACE = """ Game Keo Bua Bao

Trò chơi Kéo Búa Bao

Hãy chọn một trong ba lựa chọn dưới đây:

{% if chon %}

Bạn chọn: {{ chon.upper() }}

Máy chọn: {{ may.upper() }}

KẾT QUẢ: {{ ketqua }}

{% endif %} """ @app.route("/") def home(): return render_template_string(HTML_INTERFACE) @app.route("/play", method=["POST"]) def play(): # Lấy lựa chọn khi người dùng click vào nút trên web chon = request.form.get("luachon") # Giữ nguyên logic xử lý ngẫu nhiên từ code cũ của bạn may = random.choice(["keo", "bua", "bao"]) # Giữ nguyên logic so sánh cũ của bạn if chon == may: ketqua = "HÒA TRẬN! 🤝" elif ( (chon == "keo" and may == "bao") or (chon == "bao" and may == "bua") or (chon == "bua" and may == "keo") ): ketqua = "BẠN THẮNG RỒI! 🎉" else: ketqua = "BẠN THUA MẤT RỒI! 😢" return render_template_string( HTML_INTERFACE, chon=chon, may=may, ketqua=ketqua ) if __name__ == "__main__": app.run(debug=True)
